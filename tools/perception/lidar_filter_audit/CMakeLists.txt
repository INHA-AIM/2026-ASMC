cmake_minimum_required(VERSION 3.10)
project(lidar_filter_audit LANGUAGES CXX)
# Standalone offline harness. Does not add nodes/libraries to the catkin runtime.
set(CMAKE_CXX_STANDARD 17)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
find_package(yaml-cpp REQUIRED)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(MAIN "${REPO}/src/lidar+sensorfusion/src/Main/src")
foreach(profile planning main)
  if(profile STREQUAL main)
    set(FILTER_DIR "${MAIN}/PlannerInterface")
  else()
    set(FILTER_DIR "${REPO}/src/planning_inputs/src/filters")
  endif()
  set(FILTER_SOURCES "")
  foreach(name DrivableBoundaryFilter DrivableFootprint KnownStaticObjectFilter StaticStructureExclusionFilter)
    if(profile STREQUAL main)
      # Main's adapter header imports ROS/Global although these filters only use
      # the portable DTO. Build an exact-source snapshot with that include alone
      # narrowed to PlannerObjectTypes. Algorithm text is unchanged.
      file(READ "${FILTER_DIR}/${name}.cpp" body)
      string(REPLACE "#include <PlannerInterface/PlannerObjectAdapter.hpp>"
        "#include <PlannerInterface/PlannerObjectTypes.hpp>" body "${body}")
      file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${name}.cpp" "${body}")
      list(APPEND FILTER_SOURCES "${CMAKE_CURRENT_BINARY_DIR}/${name}.cpp")
    else()
      list(APPEND FILTER_SOURCES "${FILTER_DIR}/${name}.cpp")
    endif()
  endforeach()
  add_executable(legacy_${profile}_filter_probe legacy_filter_probe.cpp ${FILTER_SOURCES})
  target_include_directories(legacy_${profile}_filter_probe PRIVATE
    "${REPO}/src/planning_inputs/include" "${MAIN}" ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(legacy_${profile}_filter_probe PRIVATE ${OpenCV_LIBS} yaml-cpp)
  if(profile STREQUAL main)
    target_compile_definitions(legacy_${profile}_filter_probe PRIVATE USE_MAIN_FILTERS)
    target_sources(legacy_${profile}_filter_probe PRIVATE "${FILTER_DIR}/PlanningRelevanceFilter.cpp")
  endif()
endforeach()
