cmake_minimum_required(VERSION 3.16)
project(asmc_frame_time_audit LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
find_package(catkin REQUIRED COMPONENTS asmc_geometry roscpp rosbag sensor_msgs morai_msgs asmc_msgs)
find_package(Eigen3 REQUIRED)
find_package(yaml-cpp REQUIRED)
include_directories(${catkin_INCLUDE_DIRS} ${REPO}/src/asmc_lidar/include)
# Separate processes: both production estimators define the same C++ symbols.
foreach(kind bridge planning)
  if(kind STREQUAL bridge)
    set(pkg asmc_bridge)
    set(source_dir src/vehicle_state)
  else()
    set(pkg planning_inputs)
    set(source_dir src)
  endif()
  add_executable(${kind}_state_probe state_probe.cpp
    ${REPO}/src/${pkg}/${source_dir}/vehicle_state_adapter.cpp)
  target_include_directories(${kind}_state_probe PRIVATE ${REPO}/src/${pkg}/include)
  if(kind STREQUAL bridge)
    target_compile_definitions(${kind}_state_probe PRIVATE BRIDGE_PROBE)
  endif()
  target_link_libraries(${kind}_state_probe ${catkin_LIBRARIES} yaml-cpp)
endforeach()
add_library(audit_geometry STATIC ${REPO}/src/asmc_lidar/src/core/frame_time_core.cpp
  ${REPO}/src/asmc_lidar/src/geometry/projection.cpp)
target_link_libraries(audit_geometry PUBLIC Eigen3::Eigen)
add_executable(frame_projection_probe ../frame_projection_probe.cpp)
target_link_libraries(frame_projection_probe audit_geometry)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)
find_package(OpenSSL REQUIRED)
find_package(ZLIB REQUIRED)
add_executable(context_probe context_probe.cpp ${REPO}/src/asmc_lidar/src/filtering/hd_map_audit.cpp)
target_include_directories(context_probe PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(context_probe audit_geometry yaml-cpp ${OpenCV_LIBRARIES} OpenSSL::Crypto ZLIB::ZLIB)
# Runtime/support probe links already-built private LiDAR libraries only here.
# It never adds a planning dependency to the operating LiDAR target.
if(ASMC_C0_RUNTIME_LIB_DIR)
  find_package(PCL REQUIRED COMPONENTS common filters segmentation search kdtree sample_consensus)
  find_library(C0_GEOGRAPHIC_LIBRARY NAMES GeographicLib Geographic REQUIRED)
  add_executable(imu_history_probe imu_history_probe.cpp)
  add_executable(support_probe support_probe.cpp)
  target_include_directories(support_probe PRIVATE ${PCL_INCLUDE_DIRS})
  target_include_directories(imu_history_probe PRIVATE ${PCL_INCLUDE_DIRS})
  foreach(lib asmc_lidar_runtime asmc_lidar_pipeline asmc_lidar_hd_map_audit asmc_lidar_camera_core asmc_lidar_frame_time)
    find_library(C0_${lib} NAMES ${lib} PATHS ${ASMC_C0_RUNTIME_LIB_DIR} NO_DEFAULT_PATH REQUIRED)
    target_link_libraries(support_probe ${C0_${lib}})
    target_link_libraries(imu_history_probe ${C0_${lib}})
  endforeach()
  target_link_libraries(imu_history_probe ${catkin_LIBRARIES} ${PCL_LIBRARIES} ${C0_GEOGRAPHIC_LIBRARY} yaml-cpp ${OpenCV_LIBRARIES} OpenSSL::Crypto ZLIB::ZLIB)
  target_link_libraries(support_probe ${catkin_LIBRARIES} ${PCL_LIBRARIES} ${C0_GEOGRAPHIC_LIBRARY} yaml-cpp ${OpenCV_LIBRARIES} OpenSSL::Crypto ZLIB::ZLIB)
endif()

# F1 offline image overlay uses the same native camera calibration and projection.
add_executable(fusion_projection_probe fusion_projection_probe.cpp
  ${REPO}/src/asmc_lidar/src/geometry/camera_calibration.cpp)
target_link_libraries(fusion_projection_probe audit_geometry yaml-cpp OpenSSL::Crypto)
