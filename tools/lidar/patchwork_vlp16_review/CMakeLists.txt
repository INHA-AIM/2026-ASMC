cmake_minimum_required(VERSION 3.16)
project(asmc_patchwork_vlp16_review LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(H1_GENERATED_SOURCE "" CACHE PATH "Pinned generated fixed-height Patchwork++ source")
if(NOT EXISTS "${H1_GENERATED_SOURCE}/include/patchwork/patchworkpp.h")
  message(FATAL_ERROR "H1_GENERATED_SOURCE must point to the fixed-height Patchwork++ tree")
endif()
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED COMPONENTS common filters)
find_package(yaml-cpp REQUIRED)
add_executable(patchwork_vlp16_review review.cpp
  "${H1_GENERATED_SOURCE}/src/patchworkpp.cpp"
  "${H1_GENERATED_SOURCE}/src/plane_fit.cpp")
target_include_directories(patchwork_vlp16_review PRIVATE "${H1_GENERATED_SOURCE}/include" ${PCL_INCLUDE_DIRS})
target_link_libraries(patchwork_vlp16_review PRIVATE Eigen3::Eigen ${PCL_LIBRARIES} yaml-cpp pthread)
target_compile_options(patchwork_vlp16_review PRIVATE -O2 -Wall -Wextra)
