cmake_minimum_required(VERSION 3.16)
project(asmc_lidar_frame_time LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Eigen3 REQUIRED)
find_package(yaml-cpp REQUIRED)
add_library(frame_time_core core.cpp)
target_include_directories(frame_time_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(frame_time_core PUBLIC Eigen3::Eigen)
target_compile_options(frame_time_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(frame_time_eval evaluate.cpp)
target_link_libraries(frame_time_eval PRIVATE frame_time_core yaml-cpp)
add_executable(frame_time_tests test_core.cpp)
target_link_libraries(frame_time_tests PRIVATE frame_time_core)
enable_testing()
add_test(NAME frame_time_core COMMAND frame_time_tests)
