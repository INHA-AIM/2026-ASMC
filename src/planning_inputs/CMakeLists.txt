cmake_minimum_required(VERSION 3.0.2)
project(planning_inputs)
add_compile_options(-std=c++17 -Wall -Wextra)
find_package(catkin REQUIRED COMPONENTS asmc_msgs diagnostic_msgs geometry_msgs message_generation morai_msgs roscpp sensor_msgs std_msgs)
add_message_files(FILES DetectedObject.msg DetectedObjectArray.msg)
generate_messages(DEPENDENCIES geometry_msgs std_msgs)
catkin_package(INCLUDE_DIRS include CATKIN_DEPENDS asmc_msgs diagnostic_msgs geometry_msgs message_runtime morai_msgs roscpp sensor_msgs std_msgs)
include_directories(include ${catkin_INCLUDE_DIRS})
add_library(planning_state_core src/vehicle_state_adapter.cpp src/gps_transform.cpp)
add_dependencies(planning_state_core ${catkin_EXPORTED_TARGETS})
target_link_libraries(planning_state_core ${catkin_LIBRARIES})
add_executable(planning_vehicle_state_node src/vehicle_state_node.cpp)
add_dependencies(planning_vehicle_state_node ${catkin_EXPORTED_TARGETS})
target_link_libraries(planning_vehicle_state_node planning_state_core ${catkin_LIBRARIES})
add_executable(planning_objects_adapter src/objects_node.cpp src/converter.cpp)
add_dependencies(planning_objects_adapter ${${PROJECT_NAME}_EXPORTED_TARGETS} ${catkin_EXPORTED_TARGETS})
target_link_libraries(planning_objects_adapter ${catkin_LIBRARIES})
add_executable(asmc_lidar_planning_adapter src/lidar_observation_node.cpp src/converter.cpp)
add_dependencies(asmc_lidar_planning_adapter ${${PROJECT_NAME}_EXPORTED_TARGETS} ${catkin_EXPORTED_TARGETS})
target_link_libraries(asmc_lidar_planning_adapter ${catkin_LIBRARIES})
if(CATKIN_ENABLE_TESTING)
  find_package(rostest REQUIRED)
  find_package(rosbag REQUIRED)
  add_executable(planning_lidar_bag_check test/lidar_bag_check.cpp src/converter.cpp)
  target_include_directories(planning_lidar_bag_check PRIVATE ${rosbag_INCLUDE_DIRS})
  target_link_libraries(planning_lidar_bag_check ${catkin_LIBRARIES} ${rosbag_LIBRARIES})
  add_dependencies(planning_lidar_bag_check ${${PROJECT_NAME}_EXPORTED_TARGETS} ${catkin_EXPORTED_TARGETS})
  add_rostest(test/lidar_observation.test DEPENDENCIES asmc_lidar_planning_adapter)
  catkin_add_gtest(planning_lidar_companion_test test/test_lidar_planning_observation.cpp src/converter.cpp)
  if(TARGET planning_lidar_companion_test)
    add_dependencies(planning_lidar_companion_test ${${PROJECT_NAME}_EXPORTED_TARGETS} ${catkin_EXPORTED_TARGETS})
    target_link_libraries(planning_lidar_companion_test ${catkin_LIBRARIES} gtest_main)
  endif()
  foreach(test lidar_observation group_conversion)
    catkin_add_gtest(planning_${test}_test test/test_${test}.cpp src/converter.cpp)
    if(TARGET planning_${test}_test)
      add_dependencies(planning_${test}_test ${${PROJECT_NAME}_EXPORTED_TARGETS} ${catkin_EXPORTED_TARGETS})
      target_link_libraries(planning_${test}_test ${catkin_LIBRARIES})
    endif()
  endforeach()
  foreach(test grid_yaw speed_gain gps_recovery)
    catkin_add_gtest(planning_${test}_test test/test_${test}.cpp)
    if(TARGET planning_${test}_test)
      add_dependencies(planning_${test}_test ${catkin_EXPORTED_TARGETS})
      target_link_libraries(planning_${test}_test planning_state_core ${catkin_LIBRARIES} gtest_main)
    endif()
  endforeach()
endif()
install(TARGETS asmc_lidar_planning_adapter planning_state_core planning_vehicle_state_node planning_objects_adapter
  ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION}
  LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION}
  RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY include/${PROJECT_NAME}/ DESTINATION ${CATKIN_PACKAGE_INCLUDE_DESTINATION})
install(DIRECTORY launch config DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
