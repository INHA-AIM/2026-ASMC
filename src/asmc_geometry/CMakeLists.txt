cmake_minimum_required(VERSION 3.0.2)
project(asmc_geometry)
find_package(catkin REQUIRED)
catkin_package(INCLUDE_DIRS include)
install(DIRECTORY include/ DESTINATION ${CATKIN_GLOBAL_INCLUDE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  add_executable(asmc_geometry_contract_test test/contracts.cpp)
  target_include_directories(asmc_geometry_contract_test PRIVATE include)
  target_compile_options(asmc_geometry_contract_test PRIVATE -std=c++17)
  add_test(NAME asmc_geometry_contract_test COMMAND asmc_geometry_contract_test)
endif()
